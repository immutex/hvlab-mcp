1//! Bugcheck code catalog — code → symbolic name, sourced from
2//! Microsoft's public bugcheck reference. Facts only; no guidance about
3//! whether a given crash matters on any particular system.
4//!
5//! The mapping is deliberately a `match` over exact integers, never a
6//! string or substring test: a substring test like `"0x1A" in text`
7//! matches `0x1A0` and produces confident nonsense.
8
9/// A bugcheck code with its documented symbolic name and description.
10#[derive(Debug, Clone, Copy, PartialEq, Eq)]
11pub struct BugcheckInfo {
12    /// The numeric stop code.
13    pub code: u32,
14    /// The symbolic name from Microsoft's bugcheck reference.
15    pub name: &'static str,
16    /// One-line description, verbatim in spirit from the docs.
17    pub description: &'static str,
18    /// Path on `learn.microsoft.com` for the full page.
19    pub doc_path: &'static str,
20}
21
22/// Translate a bugcheck code to its documented name.
23///
24/// Unknown codes return `None`; callers decide how to present that.
25///
26/// # Examples
27///
28/// ```
29/// let info = hvlab_dump::bugcheck_name(0x9F).expect("documented code");
30/// assert_eq!(info.name, "DRIVER_POWER_STATE_FAILURE");
31/// assert!(hvlab_dump::bugcheck_name(0xDEAD_0000).is_none());
32/// ```
33#[must_use]
34pub fn bugcheck_name(code: u32) -> Option<BugcheckInfo> {
35    match code {
36        0x01 => Some(("APC_INDEX_MISMATCH", "A kernel APC index mismatch.")),
37        0x0A => Some(("IRQL_NOT_LESS_OR_EQUAL", "A kernel-mode process or driver accessed paged memory at a process IRQL that was too high.")),
38        0x1A => Some(("MEMORY_MANAGEMENT", "A severe memory management error occurred.")),
39        0x1E => Some(("KMODE_EXCEPTION_NOT_HANDLED", "A kernel-mode program generated an exception the error handler did not catch.")),
40        0x24 => Some(("NTFS_FILE_SYSTEM", "A problem occurred within the NTFS file system driver.")),
41        0x3B => Some(("SYSTEM_SERVICE_EXCEPTION", "An exception happened while executing a routine that transitions from non-privileged code to privileged code.")),
42        0x44 => Some(("MULTIPLE_IRP_COMPLETE_REQUESTS", "A driver has called `IoCompleteRequest` on an IRP already completed.")),
43        0x50 => Some(("PAGE_FAULT_IN_NONPAGED_AREA", "Invalid system memory was referenced, typically in the non-paged pool.")),
44        0x7B => Some(("INACCESSIBLE_BOOT_DEVICE", "The OS lost access to the system partition during boot.")),
45        0x7E => Some(("SYSTEM_THREAD_EXCEPTION_NOT_HANDLED", "A system thread generated an exception the error handler did not catch.")),
46        0x7F => Some(("UNEXPECTED_KERNEL_MODE_TRAP", "The kernel ran into an unexpected trap or double fault.")),
47        0x8E => Some(("KERNEL_MODE_EXCEPTION_NOT_HANDLED", "A kernel-mode program generated an exception the error handler did not catch.")),
48        0x9F => Some(("DRIVER_POWER_STATE_FAILURE", "A driver is in an inconsistent or invalid power state.")),
49        0xA0 => Some(("INTERNAL_POWER_ERROR", "The power policy manager experienced a severe error.")),
50        0xC2 => Some(("BAD_POOL_CALLER", "A pool header was corrupted, typically by a bad pool allocation or free.")),
51        0xC5 => Some(("DRIVER_CORRUPTED_EXPOOL", "A driver attempted to access pageable memory at a too-high process IRQL.")),
52        0xD1 => Some(("DRIVER_IRQL_NOT_LESS_OR_EQUAL", "A driver attempted to access pageable memory at a too-high process IRQL.")),
53        0x101 => Some(("CLOCK_WATCHDOG_TIMEOUT", "An expected clock interrupt was not received on a secondary processor in an SMP system.")),
54        0x109 => Some(("CRITICAL_STRUCTURE_CORRUPTION", "The kernel has detected the corruption of a critical data structure.")),
55        0x116 => Some(("VIDEO_TDR_FAILURE", "The display driver failed to complete a timed recovery operation.")),
56        0x124 => Some(("WHEA_UNCORRECTABLE_ERROR", "Windows Hardware Error Architecture reported an uncorrectable hardware error.")),
57        0x133 => Some(("DPC_WATCHDOG_VIOLATION", "The DPC watchdog detected a prolonged run at DISPATCH_LEVEL or above.")),
58        0x139 => Some(("KERNEL_SECURITY_CHECK_FAILURE", "The kernel detected the corruption of a critical data structure (fast fail).")),
59        0x18B => Some(("SECURE_KERNEL_ERROR", "The secure kernel encountered an error condition.")),
60        0x1A0 => Some(("TTM_FATAL_ERROR", "The video memory manager encountered a fatal error.")),
61        0x1C7 => Some(("STORE_DATA_STRUCTURE_CORRUPTION", "A data structure in the memory store is corrupted.")),
62        _ => None,
63    }
64    .map(|(name, description)| BugcheckInfo {
65        code,
66        name,
67        description,
68        doc_path: "https://learn.microsoft.com/en-us/windows-hardware/drivers/debugger/bug-check-code-reference2",
69    })
70}

72/// Format a bugcheck code the way `WinDbg` prints it: `0x0000009F`.
73#[must_use]
74pub fn format_bugcheck_code(code: u32) -> String {
75    format!("0x{code:08X}")
76}

78#[cfg(test)]
79mod tests {
80    use super::*;
81
82    mod bugcheck_name {
83        use super::*;
84
85        #[test]
86        fn returns_documented_name_for_known_code() {
87            let info = bugcheck_name(0x9F).expect("0x9F is documented");
88            assert_eq!(info.name, "DRIVER_POWER_STATE_FAILURE");
89            assert_eq!(info.code, 0x9F);
90        }
91
92        #[test]
93        fn returns_none_for_undocumented_code() {
94            assert!(bugcheck_name(0xDEAD_0000).is_none());
95        }
96
97        #[test]
98        fn exact_codes_only_short_code_never_matches_longer_one() {
99            // The classic hazard: a substring test for "0x1A" also
100            // matches 0x1A0. Integer matching cannot make that mistake.
101            assert_eq!(bugcheck_name(0x1A).map(|i| i.name), Some("MEMORY_MANAGEMENT"));
102            assert_eq!(bugcheck_name(0x1A0).map(|i| i.name), Some("TTM_FATAL_ERROR"));
103            assert_ne!(
104                bugcheck_name(0x1A).map(|i| i.name),
105                bugcheck_name(0x1A0).map(|i| i.name)
106            );
107        }
108
109        #[test]
110        fn adjacent_codes_do_not_alias() {
111            assert_eq!(bugcheck_name(0x50).map(|i| i.name), Some("PAGE_FAULT_IN_NONPAGED_AREA"));
112            assert_eq!(bugcheck_name(0x500).map(|i| i.name), None);
113            assert_eq!(bugcheck_name(0x7E).map(|i| i.name), Some("SYSTEM_THREAD_EXCEPTION_NOT_HANDLED"));
114            assert_eq!(bugcheck_name(0x7E0).map(|i| i.name), None);
115        }
116    }
117
118    mod format_bugcheck_code {
119        use super::*;
120
121        #[test]
122        fn pads_to_eight_digits() {
123            assert_eq!(format_bugcheck_code(0x9F), "0x0000009F");
124            assert_eq!(format_bugcheck_code(0x1A), "0x0000001A");
125        }
126
127        #[test]
128        fn uppercase_hex() {
129            assert_eq!(format_bugcheck_code(0xAB), "0x000000AB");
130        }
131    }
